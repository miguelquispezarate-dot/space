        if (!this.enabled) return;
        this.initContext();
        if (!this.ctx) return;
        try {
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.type = 'sine';
          osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
          osc.frequency.exponentialRampToValueAtTime(freq * 1.3, this.ctx.currentTime + duration);
          gain.gain.setValueAtTime(0.06, this.ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
          osc.connect(gain);
          gain.connect(this.ctx.destination);
          osc.start();
          osc.stop(this.ctx.currentTime + duration);
        } catch(e) {}
      }

        if (!this.enabled) return;
        this.initContext();
        if (!this.ctx) return;
        try {
          const bufferSize = this.ctx.sampleRate * duration;
          const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
          const data = buffer.getChannelData(0);
          let lastVal = 0.0;
          for (let i = 0; i < bufferSize; i++) {
            const white = Math.random() * 2 - 1;
            lastVal = (lastVal + (0.05 * white)) / 1.05;
            data[i] = lastVal * 2.8;
          }

          const noiseSource = this.ctx.createBufferSource();
          noiseSource.buffer = buffer;

          const filter = this.ctx.createBiquadFilter();
          filter.type = 'lowpass';
          filter.frequency.setValueAtTime(150, this.ctx.currentTime);
          filter.frequency.exponentialRampToValueAtTime(800, this.ctx.currentTime + duration * 0.7);

          const gain = this.ctx.createGain();
          gain.gain.setValueAtTime(0.06, this.ctx.currentTime);
          gain.gain.linearRampToValueAtTime(0.38, this.ctx.currentTime + 1.2);
          gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + duration);

          noiseSource.connect(filter);
          filter.connect(gain);
          gain.connect(this.ctx.destination);

          noiseSource.start();
          noiseSource.stop(this.ctx.currentTime + duration);
        } catch(e) {}
      }

    class NasaSpaceApp {
      constructor() {
        this.currentScreen = 'launch';
        this.selectedCelestial = 'mars';
        this.currentSurface = null;

        // DOM Elements
        this.screenLaunch = document.getElementById('screen-launch');
        this.screenSolar = document.getElementById('screen-solar');
        this.screenSurface = document.getElementById('screen-surface');
        this.labelHudStatus = document.getElementById('label-hud-status');
        this.btnBackOrbit = document.getElementById('btn-back-orbit');
        this.btnAudioMute = document.getElementById('btn-audio-mute');
        this.iconSoundOn = document.getElementById('icon-sound-on');
        this.iconSoundOff = document.getElementById('icon-sound-off');

        // Encyclopedia Module
        this.encyclopediaDrawer = document.getElementById('encyclopedia-drawer');
        this.btnToggleEnc = document.getElementById('btn-toggle-enc');
        this.btnCloseEnc = document.getElementById('btn-close-enc');
        this.encArticlesScroll = document.getElementById('enc-articles-scroll');

        // Graphics Views
        this.launchView = new NasaLaunchScene('launch-canvas');
        this.solarView = new NasaSolarOrrery('solar-canvas');
        this.surfaceView = new NasaSurfaceLandscape();

        this.bindEvents();
        this.renderEncyclopediaTab('apollo');
      }

      bindEvents() {
        // Audio Toggle
        this.btnAudioMute.addEventListener('click', () => {
          const state = sound.toggleAudio();
          this.iconSoundOn.style.display = state ? 'block' : 'none';
          this.iconSoundOff.style.display = state ? 'none' : 'block';
          if (state) sound.playTelemetryChirp(1000);
        });

        // Liftoff Button
        document.getElementById('btn-start-liftoff').addEventListener('click', () => {
          this.executeLiftoff();
        });

        // Return to Orbit Button
        this.btnBackOrbit.addEventListener('click', () => {
          sound.playTelemetryChirp(700);
          this.switchView('solar');
        });

        // Solar System Navigation Chips
        document.querySelectorAll('#solar-chips-nav .body-nav-chip').forEach(chip => {
          chip.addEventListener('click', () => {
            document.querySelectorAll('#solar-chips-nav .body-nav-chip').forEach(c => c.classList.remove('active'));
            chip.classList.add('active');
            const targetId = chip.dataset.id;
            sound.playTelemetryChirp(820);
            this.solarView.focusOnTarget(targetId);
          });
        });

        // Zoom Controls
        document.getElementById('btn-zoom-plus').addEventListener('click', () => {
          sound.playTelemetryChirp(900);
          this.solarView.adjustZoom(1.28);
        });
        document.getElementById('btn-zoom-minus').addEventListener('click', () => {
          sound.playTelemetryChirp(700);
          this.solarView.adjustZoom(0.78);
        });

        // Dive into Surface Button
        document.getElementById('btn-dive-surface').addEventListener('click', () => {
          sound.playTelemetryChirp(1100);
          this.diveIntoSurface(this.selectedCelestial);
        });

        // Close Celestial Drawer
        document.getElementById('drawer-close-button').addEventListener('click', () => {
          sound.playTelemetryChirp(550);
          document.getElementById('celestial-drawer').classList.remove('open');
        });

        // Close Inspection Modal
        document.getElementById('btn-close-modal').addEventListener('click', () => {
          sound.playTelemetryChirp(550);
          document.getElementById('inspection-modal').classList.remove('open');
        });

        // Encyclopedia Toggle
        this.btnToggleEnc.addEventListener('click', () => {
          sound.playTelemetryChirp(850);
          this.encyclopediaDrawer.classList.toggle('open');
        });
        this.btnCloseEnc.addEventListener('click', () => {
          sound.playTelemetryChirp(550);
          this.encyclopediaDrawer.classList.remove('open');
        });

        // Encyclopedia Tab Chips
        document.querySelectorAll('.enc-tab-chip').forEach(btn => {
          btn.addEventListener('click', () => {
            document.querySelectorAll('.enc-tab-chip').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            sound.playTelemetryChirp(780);
            this.renderEncyclopediaTab(btn.dataset.tab);
          });
        });

        // Responsive Window Resize
        window.addEventListener('resize', () => {
          this.launchView.handleResize();
          this.solarView.handleResize();
        });
      }

      switchView(screenKey) {
        this.currentScreen = screenKey;
        [this.screenLaunch, this.screenSolar, this.screenSurface].forEach(s => s.classList.remove('active'));

        if (screenKey === 'launch') {
          this.screenLaunch.classList.add('active');
          this.btnBackOrbit.style.display = 'none';
          this.labelHudStatus.innerText = 'CAPE CANAVERAL // T-0 STANDBY';
        } else if (screenKey === 'solar') {
          this.screenSolar.classList.add('active');
          this.btnBackOrbit.style.display = 'none';
          this.labelHudStatus.innerText = 'NASA ORBITAL RECON // SOLAR SYSTEM';
          this.solarView.startLoop();
        } else if (screenKey === 'surface') {
          this.screenSurface.classList.add('active');
          this.btnBackOrbit.style.display = 'inline-flex';
          const name = this.currentSurface === 'mars' ? 'MARS // JEZERO CRATER (JPL)' : 'THE MOON // HADLEY-APENNINE (APOLLO 15)';
          this.labelHudStatus.innerText = `SURFACE RECON // ${name}`;
        }
      }

      executeLiftoff() {
        sound.initContext();
        sound.playRocketIgnition(4.5);

        if (navigator.vibrate) {
          navigator.vibrate([120, 60, 250, 60, 450]);
        }

        this.screenLaunch.classList.add('camera-shake-active');
        this.labelHudStatus.innerText = 'LIFTOFF // SLS MAIN IGNITION';

        this.launchView.startLiftoff(() => {
          this.screenLaunch.classList.remove('camera-shake-active');
          this.switchView('solar');
          setTimeout(() => {
            this.solarView.selectBody('earth');
          }, 800);
        });
      }

      openBodyDetails(key) {
        this.selectedCelestial = key;
        const body = NASA_CELESTIAL_BODIES[key];
        if (!body) return;

        sound.playBeaconPing();
        document.getElementById('drawer-tag').innerText = body.tag;
        document.getElementById('drawer-name').innerText = body.name;
        document.getElementById('stat-dist').innerText = body.dist;
        document.getElementById('stat-grav').innerText = body.grav;
        document.getElementById('stat-diam').innerText = body.diam;
        document.getElementById('drawer-narrative').innerText = body.narrative;

        const missionsCont = document.getElementById('drawer-missions-row');
        missionsCont.innerHTML = '';
        body.missions.forEach(m => {
          const span = document.createElement('span');
          span.className = 'mission-pill';
          span.innerText = m;
          missionsCont.appendChild(span);
        });

        const diveBtn = document.getElementById('btn-dive-surface');
        if (body.hasSurface) {
          diveBtn.style.display = 'inline-flex';
          diveBtn.querySelector('span').innerText = `EXPLORE SURFACE OF ${body.name.toUpperCase()}`;
        } else {
          diveBtn.style.display = 'none';
        }

        document.getElementById('celestial-drawer').classList.add('open');
      }

      diveIntoSurface(bodyKey) {
        document.getElementById('celestial-drawer').classList.remove('open');
        this.currentSurface = bodyKey === 'moon' ? 'moon' : 'mars';
        this.surfaceView.setupScenario(this.currentSurface);
        this.switchView('surface');
      }

      renderEncyclopediaTab(tabKey) {
        const articles = NASA_ENCYCLOPEDIA_ARTICLES[tabKey] || [];
        this.encArticlesScroll.innerHTML = '';

        articles.forEach(art => {
          const card = document.createElement('div');
          card.className = 'enc-card';

          let tableHtml = '';
          if (art.table && art.table.length > 0) {
            tableHtml = '<table class="enc-table">';
            art.table.forEach(([k, v]) => {
              tableHtml += `<tr><td>${k}</td><td>${v}</td></tr>`;
            });
            tableHtml += '</table>';
          }

          card.innerHTML = `
            <div class="tag">${art.tag}</div>
            <h4>${art.title}</h4>
            <p>${art.text}</p>
            ${tableHtml}
          `;

    class NasaLaunchScene {
      constructor(canvasId) {
        this.canvas = document.getElementById(canvasId);
        this.ctx = this.canvas.getContext('2d');
        this.particles = [];
        this.stars = [];
        this.rocketY = 0;
        this.rocketV = 0;
        this.isLaunching = false;
        this.altKm = 0;
        this.onFinishCallback = null;

        this.handleResize();
        this.initStars();
        this.renderLoop();
      }

      handleResize() {
        this.canvas.width = window.innerWidth;
        this.canvas.height = window.innerHeight;
        if (!this.isLaunching) {
          this.rocketY = this.canvas.height * 0.63;
        }
      }

      initStars() {
        this.stars = [];
        for (let i = 0; i < 150; i++) {
          this.stars.push({
            x: Math.random() * window.innerWidth,
            y: Math.random() * window.innerHeight,
            size: Math.random() * 1.8 + 0.5,
            alpha: Math.random() * 0.8 + 0.2
          });
        }
      }

      startLiftoff(callback) {
        this.isLaunching = true;
        this.rocketV = 0.9;
        this.onFinishCallback = callback;
      }

      renderLoop() {
        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

        // Atmospheric to space color transition
        const progress = Math.min(this.altKm / 120, 1);
        const skyGradient = this.ctx.createLinearGradient(0, 0, 0, this.canvas.height);

        if (progress < 0.5) {
          skyGradient.addColorStop(0, '#060d21');
          skyGradient.addColorStop(0.7, '#132145');
          skyGradient.addColorStop(1, '#2c1e28');
        } else {
          skyGradient.addColorStop(0, '#020409');
          skyGradient.addColorStop(1, '#091329');
        }
        this.ctx.fillStyle = skyGradient;
        this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);

        // Stars
        this.stars.forEach(s => {
          this.ctx.fillStyle = `rgba(255, 255, 255, ${s.alpha * (0.3 + progress * 0.7)})`;
          this.ctx.beginPath();
          this.ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
          this.ctx.fill();
        });

        // Launchpad and gantry service tower
        const groundY = this.canvas.height * 0.76 + (this.isLaunching ? this.altKm * 5.5 : 0);
        if (groundY < this.canvas.height + 220) {
          this.drawPad(groundY);
        }

        // Volumetric smoke and fire plumes
        if (this.isLaunching || Math.random() < 0.25) {
          const count = this.isLaunching ? 9 : 2;
          for (let i = 0; i < count; i++) {
            this.particles.push({
              x: this.canvas.width / 2 + (Math.random() - 0.5) * 32,
              y: this.rocketY + 92,
              vx: (Math.random() - 0.5) * (this.isLaunching ? 7 : 1.5),
              vy: Math.random() * (this.isLaunching ? 8 : 2) + 5,
              radius: Math.random() * (this.isLaunching ? 20 : 6) + 5,
              type: Math.random() > 0.4 ? 'fire' : 'smoke',
              life: 1,
              decay: Math.random() * 0.035 + 0.02
            });
          }
        }

        for (let i = this.particles.length - 1; i >= 0; i--) {
          const p = this.particles[i];
          p.x += p.vx;
          p.y += p.vy;
          p.radius += 0.5;
          p.life -= p.decay;

          if (p.life <= 0) {
            this.particles.splice(i, 1);
            continue;
          }

          if (p.type === 'fire') {
            this.ctx.fillStyle = `rgba(245, 158, 11, ${p.life * 0.85})`;
          } else {
            this.ctx.fillStyle = `rgba(148, 163, 184, ${p.life * 0.5})`;
          }
          this.ctx.beginPath();
          this.ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
          this.ctx.fill();
        }

        // Draw NASA Space Launch System (SLS)
        this.drawNasaSLS(this.canvas.width / 2, this.rocketY);

        if (this.isLaunching) {
          this.rocketV += 0.17;
          this.rocketY -= this.rocketV;
          this.altKm += this.rocketV * 0.4;

          if (this.rocketY < -240 && this.onFinishCallback) {
            const cb = this.onFinishCallback;
            this.onFinishCallback = null;
            setTimeout(() => cb(), 300);
          }
        }

        requestAnimationFrame(() => this.renderLoop());
      }

      drawPad(y) {
        const cx = this.canvas.width / 2;
        this.ctx.fillStyle = '#1e2638';
        this.ctx.fillRect(cx - 120, y, 240, 80);

        this.ctx.strokeStyle = '#475569';
        this.ctx.lineWidth = 3;
        this.ctx.strokeRect(cx + 45, y - 200, 45, 200);

        for (let gy = y - 180; gy < y; gy += 30) {
          this.ctx.beginPath();
          this.ctx.moveTo(cx + 45, gy);
          this.ctx.lineTo(cx + 90, gy + 30);
          this.ctx.stroke();
        }

        this.ctx.fillStyle = '#f59e0b';
        this.ctx.beginPath();
        this.ctx.arc(cx - 100, y + 10, 4, 0, Math.PI * 2);
        this.ctx.arc(cx + 100, y + 10, 4, 0, Math.PI * 2);
        this.ctx.fill();
      }

      drawNasaSLS(x, y) {
        this.ctx.save();
        this.ctx.translate(x, y);

        // Core Stage (Cryogenic orange insulation foam)
        this.ctx.fillStyle = '#ea580c';
        this.ctx.fillRect(-14, -20, 28, 95);

        // Inter-stage white rings
        this.ctx.fillStyle = '#f8fafc';
        this.ctx.fillRect(-14, 0, 28, 8);
        this.ctx.fillRect(-14, 40, 28, 6);

        // Orion capsule & Launch Abort System (LAS) tower
        this.ctx.fillStyle = '#f8fafc';
        this.ctx.beginPath();
        this.ctx.moveTo(0, -75);
        this.ctx.lineTo(2, -45);
        this.ctx.lineTo(12, -20);
        this.ctx.lineTo(-12, -20);
        this.ctx.lineTo(-2, -45);
        this.ctx.closePath();
        this.ctx.fill();

        // Blue NASA insignia emblem
        this.ctx.fillStyle = '#0b3d91';
        this.ctx.beginPath();
        this.ctx.arc(0, 20, 4.5, 0, Math.PI * 2);
        this.ctx.fill();

        // Twin Solid Rocket Boosters (SRBs)
        const drawBooster = (bx) => {
          this.ctx.fillStyle = '#f1f5f9';
          this.ctx.beginPath();
          this.ctx.moveTo(bx, -25);
          this.ctx.lineTo(bx + 7, -10);
          this.ctx.lineTo(bx + 7, 75);
          this.ctx.lineTo(bx - 7, 75);
          this.ctx.lineTo(bx - 7, -10);
          this.ctx.closePath();
          this.ctx.fill();

          this.ctx.fillStyle = '#0f172a';
          this.ctx.fillRect(bx - 7, 10, 14, 3);
          this.ctx.fillRect(bx - 7, 40, 14, 3);
        };

        drawBooster(-21);
        drawBooster(21);

        // 4x RS-25 engine cluster
        this.ctx.fillStyle = '#334155';
        this.ctx.fillRect(-10, 75, 20, 10);

        // Exhaust thrust fire
        if (this.isLaunching) {
          this.ctx.fillStyle = '#fbbf24';
          this.ctx.beginPath();
          this.ctx.moveTo(-10, 85);
          this.ctx.lineTo(0, 145 + Math.random() * 25);
          this.ctx.lineTo(10, 85);
          this.ctx.closePath();
          this.ctx.fill();

          const drawBoosterFlame = (bx) => {
            this.ctx.fillStyle = '#f59e0b';
            this.ctx.beginPath();
            this.ctx.moveTo(bx - 5, 75);
            this.ctx.lineTo(bx, 115 + Math.random() * 15);
            this.ctx.lineTo(bx + 5, 75);
            this.ctx.closePath();
            this.ctx.fill();
          };
          drawBoosterFlame(-21);
          drawBoosterFlame(21);
        }

        this.ctx.restore();
      }
    }

       H. SCREEN 2: PHOTOREALISTIC SOLAR SYSTEM ORRERY

    class NasaSolarOrrery {
      constructor(canvasId) {
        this.canvas = document.getElementById(canvasId);
        this.ctx = this.canvas.getContext('2d');
        this.active = false;

        this.cam = { x: 0, y: 0, scale: 0.95 };
        this.dragging = false;
        this.dragOrigin = { x: 0, y: 0 };
        this.lastTouchDist = 0;
        this.orbitTime = 0;

        this.backgroundStars = [];
        this.asteroidBelt = [];

        this.handleResize();
        this.initStarfield();
        this.bindUserGestures();
      }

      handleResize() {
        this.canvas.width = window.innerWidth;
        this.canvas.height = window.innerHeight;
        this.cam.x = this.canvas.width / 2;
        this.cam.y = this.canvas.height / 2;
      }

      initStarfield() {
        this.backgroundStars = [];
        for (let i = 0; i < 340; i++) {
          this.backgroundStars.push({
            x: (Math.random() - 0.5) * 3500,
            y: (Math.random() - 0.5) * 3500,
            r: Math.random() * 1.6 + 0.4,
            a: Math.random() * 0.7 + 0.3
          });
        }

        this.asteroidBelt = [];
        for (let i = 0; i < 220; i++) {
          const angle = Math.random() * Math.PI * 2;
          const dist = 275 + (Math.random() - 0.5) * 48;
          this.asteroidBelt.push({
            angle: angle,
            dist: dist,
            speed: (Math.random() * 0.002 + 0.001),
            size: Math.random() * 2.2 + 0.8
          });
        }
      }

      bindUserGestures() {
        this.canvas.addEventListener('mousedown', (e) => {
          this.dragging = true;
          this.dragOrigin = { x: e.clientX - this.cam.x, y: e.clientY - this.cam.y };
        });

        window.addEventListener('mousemove', (e) => {
          if (!this.dragging) return;
          this.cam.x = e.clientX - this.dragOrigin.x;
          this.cam.y = e.clientY - this.dragOrigin.y;
        });

        window.addEventListener('mouseup', () => {
          this.dragging = false;
        });

        this.canvas.addEventListener('wheel', (e) => {
          e.preventDefault();
          const zoomRatio = e.deltaY < 0 ? 1.15 : 0.88;
          this.adjustZoom(zoomRatio);
        }, { passive: false });

        this.canvas.addEventListener('touchstart', (e) => {
          if (e.touches.length === 1) {
            this.dragging = true;
            this.dragOrigin = {
              x: e.touches[0].clientX - this.cam.x,
              y: e.touches[0].clientY - this.cam.y
            };
          } else if (e.touches.length === 2) {
            this.dragging = false;
            this.lastTouchDist = Math.hypot(
              e.touches[0].clientX - e.touches[1].clientX,
              e.touches[0].clientY - e.touches[1].clientY
            );
          }
        });

        this.canvas.addEventListener('touchmove', (e) => {
          if (e.touches.length === 1 && this.dragging) {
            this.cam.x = e.touches[0].clientX - this.dragOrigin.x;
            this.cam.y = e.touches[0].clientY - this.dragOrigin.y;
          } else if (e.touches.length === 2) {
            const currentDist = Math.hypot(
              e.touches[0].clientX - e.touches[1].clientX,
              e.touches[0].clientY - e.touches[1].clientY
            );
            if (this.lastTouchDist > 0) {
              const factor = currentDist / this.lastTouchDist;
              this.adjustZoom(factor);
            }
            this.lastTouchDist = currentDist;
          }
        }, { passive: true });

        this.canvas.addEventListener('touchend', () => {
          this.dragging = false;
          this.lastTouchDist = 0;
        });

        this.canvas.addEventListener('click', (e) => {
          this.resolveRaycast(e.clientX, e.clientY);
        });
      }

      adjustZoom(factor) {
        this.cam.scale = Math.min(Math.max(this.cam.scale * factor, 0.35), 3.4);
      }

      focusOnTarget(key) {
        if (key === 'all') {
          this.cam.x = this.canvas.width / 2;
          this.cam.y = this.canvas.height / 2;
          this.cam.scale = 0.95;
          return;
        }

        const body = NASA_CELESTIAL_BODIES[key];
        if (!body) return;

        const angle = this.orbitTime * body.speed;
        const bx = Math.cos(angle) * body.orbitRadius;
        const by = Math.sin(angle) * body.orbitRadius;

        this.cam.x = this.canvas.width / 2 - bx * 1.5;
        this.cam.y = this.canvas.height / 2 - by * 1.5;
        this.cam.scale = 1.65;

        app.openBodyDetails(key);
      }

      selectBody(key) {
        this.focusOnTarget(key);
      }

      resolveRaycast(screenX, screenY) {
        const worldX = (screenX - this.cam.x) / this.cam.scale;
        const worldY = (screenY - this.cam.y) / this.cam.scale;

        for (const [key, body] of Object.entries(NASA_CELESTIAL_BODIES)) {
          let bx = 0;
          let by = 0;
          if (body.orbitRadius > 0) {
            const angle = this.orbitTime * body.speed;
            bx = Math.cos(angle) * body.orbitRadius;
            by = Math.sin(angle) * body.orbitRadius;
          }

          const dist = Math.hypot(worldX - bx, worldY - by);
          const hitRadius = Math.max(body.radius * 1.6, 26);

          if (dist <= hitRadius) {
            app.openBodyDetails(key);
            return;
          }
        }
      }

      startLoop() {
        if (!this.active) {
          this.active = true;
          this.drawFrame();
        }
      }

      drawFrame() {
        if (!this.active) return;
        this.orbitTime += 1;

        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

        // Void background
        this.ctx.fillStyle = '#010308';
        this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);

        this.ctx.save();
        this.ctx.translate(this.cam.x, this.cam.y);
        this.ctx.scale(this.cam.scale, this.cam.scale);

        // Starfield
        this.backgroundStars.forEach(s => {
          this.ctx.fillStyle = `rgba(255, 255, 255, ${s.a})`;
          this.ctx.beginPath();
          this.ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
          this.ctx.fill();
        });

        // Orbital ellipses
        for (const [key, body] of Object.entries(NASA_CELESTIAL_BODIES)) {
          if (body.orbitRadius > 0) {
            this.ctx.strokeStyle = 'rgba(148, 163, 184, 0.16)';
            this.ctx.lineWidth = 1 / this.cam.scale;
            this.ctx.setLineDash([4, 4]);
            this.ctx.beginPath();
            this.ctx.arc(0, 0, body.orbitRadius, 0, Math.PI * 2);
            this.ctx.stroke();
            this.ctx.setLineDash([]);
          }
        }

        // Asteroid belt
        this.ctx.fillStyle = 'rgba(148, 163, 184, 0.55)';
        this.asteroidBelt.forEach(ast => {
          ast.angle += ast.speed;
          const ax = Math.cos(ast.angle) * ast.dist;
          const ay = Math.sin(ast.angle) * ast.dist;
          this.ctx.beginPath();
          this.ctx.arc(ax, ay, ast.size, 0, Math.PI * 2);
          this.ctx.fill();
        });

        // Draw planets with 3D sphere shading
        for (const [key, body] of Object.entries(NASA_CELESTIAL_BODIES)) {
          let bx = 0;
          let by = 0;
          if (body.orbitRadius > 0) {
            const angle = this.orbitTime * body.speed;
            bx = Math.cos(angle) * body.orbitRadius;
            by = Math.sin(angle) * body.orbitRadius;
          }

          this.drawPhotorealisticPlanet(bx, by, body);

          // Clean NASA typography label
          this.ctx.fillStyle = '#cbd5e1';
          this.ctx.font = '10px monospace';
          this.ctx.textAlign = 'center';
          this.ctx.fillText(body.name.toUpperCase(), bx, by + body.radius + 15);
        }

        this.ctx.restore();
        requestAnimationFrame(() => this.drawFrame());
      }

      drawPhotorealisticPlanet(bx, by, body) {
        this.ctx.save();
        this.ctx.translate(bx, by);

        const r = body.radius;

        if (body.textureType === 'sun') {
          const corona = this.ctx.createRadialGradient(0, 0, r * 0.4, 0, 0, r * 2.2);
          corona.addColorStop(0, '#fffbeb');
          corona.addColorStop(0.3, '#f59e0b');
          corona.addColorStop(0.8, 'rgba(217, 119, 6, 0.25)');
          corona.addColorStop(1, 'rgba(0,0,0,0)');
          this.ctx.fillStyle = corona;
          this.ctx.beginPath();
          this.ctx.arc(0, 0, r * 2.2, 0, Math.PI * 2);
          this.ctx.fill();

          this.ctx.fillStyle = '#fbbf24';
          this.ctx.beginPath();
          this.ctx.arc(0, 0, r, 0, Math.PI * 2);
          this.ctx.fill();

        } else if (body.textureType === 'earth') {
          const sphereGrad = this.ctx.createRadialGradient(-r * 0.3, -r * 0.3, r * 0.1, 0, 0, r);
          sphereGrad.addColorStop(0, '#60a5fa');
          sphereGrad.addColorStop(0.5, '#1d4ed8');
          sphereGrad.addColorStop(1, '#0f172a');
          this.ctx.fillStyle = sphereGrad;
          this.ctx.beginPath();
          this.ctx.arc(0, 0, r, 0, Math.PI * 2);
          this.ctx.fill();

          this.ctx.fillStyle = '#15803d';
          this.ctx.beginPath();
          this.ctx.arc(-2, -1, r * 0.5, 0, Math.PI * 1.4);
          this.ctx.fill();

          this.ctx.fillStyle = 'rgba(255, 255, 255, 0.45)';
          this.ctx.beginPath();
          this.ctx.ellipse(1, -2, r * 0.7, r * 0.25, 0.4, 0, Math.PI * 2);
          this.ctx.fill();

          this.ctx.strokeStyle = 'rgba(56, 189, 248, 0.6)';
          this.ctx.lineWidth = 1.5;
          this.ctx.stroke();

        } else if (body.textureType === 'moon') {
          const moonGrad = this.ctx.createRadialGradient(-r * 0.3, -r * 0.3, r * 0.1, 0, 0, r);
          moonGrad.addColorStop(0, '#f1f5f9');
          moonGrad.addColorStop(0.7, '#64748b');
          moonGrad.addColorStop(1, '#1e293b');
          this.ctx.fillStyle = moonGrad;
          this.ctx.beginPath();
          this.ctx.arc(0, 0, r, 0, Math.PI * 2);
          this.ctx.fill();

          this.ctx.fillStyle = 'rgba(30, 41, 59, 0.65)';
          this.ctx.beginPath();
          this.ctx.arc(-1, -1, r * 0.35, 0, Math.PI * 2);
          this.ctx.arc(2, 2, r * 0.25, 0, Math.PI * 2);
          this.ctx.fill();

        } else if (body.textureType === 'mars') {
          const marsGrad = this.ctx.createRadialGradient(-r * 0.3, -r * 0.3, r * 0.1, 0, 0, r);
          marsGrad.addColorStop(0, '#f97316');
          marsGrad.addColorStop(0.6, '#c2410c');
          marsGrad.addColorStop(1, '#431407');
          this.ctx.fillStyle = marsGrad;
          this.ctx.beginPath();
          this.ctx.arc(0, 0, r, 0, Math.PI * 2);
          this.ctx.fill();

          this.ctx.fillStyle = '#f8fafc';
          this.ctx.beginPath();
          this.ctx.ellipse(0, -r + 2, r * 0.4, r * 0.18, 0, 0, Math.PI * 2);
          this.ctx.fill();

          this.ctx.strokeStyle = '#290e05';
          this.ctx.lineWidth = 1.8;
          this.ctx.beginPath();
          this.ctx.moveTo(-r * 0.6, 1);
          this.ctx.lineTo(r * 0.5, 3);
          this.ctx.stroke();

        } else if (body.textureType === 'jupiter') {
          const jupGrad = this.ctx.createRadialGradient(-r * 0.3, -r * 0.3, r * 0.1, 0, 0, r);
          jupGrad.addColorStop(0, '#fed7aa');
          jupGrad.addColorStop(0.7, '#c2410c');
          jupGrad.addColorStop(1, '#451a03');
          this.ctx.fillStyle = jupGrad;
          this.ctx.beginPath();
          this.ctx.arc(0, 0, r, 0, Math.PI * 2);
          this.ctx.fill();

          this.ctx.strokeStyle = 'rgba(124, 45, 18, 0.7)';
          this.ctx.lineWidth = 2.5;
          this.ctx.beginPath();
          this.ctx.moveTo(-r * 0.85, -6);
          this.ctx.lineTo(r * 0.85, -6);
          this.ctx.moveTo(-r * 0.95, 2);
          this.ctx.lineTo(r * 0.95, 2);
          this.ctx.moveTo(-r * 0.85, 9);
          this.ctx.lineTo(r * 0.85, 9);
          this.ctx.stroke();

          this.ctx.fillStyle = '#b91c1c';
          this.ctx.beginPath();
          this.ctx.ellipse(6, 6, 4.5, 3, 0.1, 0, Math.PI * 2);
          this.ctx.fill();

        } else {
          const genGrad = this.ctx.createRadialGradient(-r * 0.3, -r * 0.3, r * 0.1, 0, 0, r);
          genGrad.addColorStop(0, body.color);
          genGrad.addColorStop(1, '#0f172a');
          this.ctx.fillStyle = genGrad;
          this.ctx.beginPath();
          this.ctx.arc(0, 0, r, 0, Math.PI * 2);
          this.ctx.fill();
        }

        this.ctx.restore();
      }
    }

      initPanGestures() {
        this.viewport.addEventListener('mousedown', (e) => {
          this.isPanning = true;
          this.panTouchStartX = e.clientX;
          this.panStartOffset = this.panOffsetX;
        });

        window.addEventListener('mousemove', (e) => {
          if (!this.isPanning) return;
          const delta = e.clientX - this.panTouchStartX;
          this.setPan(this.panStartOffset + delta);
        });

        window.addEventListener('mouseup', () => {
          this.isPanning = false;
        });

        this.viewport.addEventListener('touchstart', (e) => {
          if (e.touches.length === 1) {
            this.isPanning = true;
            this.panTouchStartX = e.touches[0].clientX;
            this.panStartOffset = this.panOffsetX;
          }
        }, { passive: true });

        this.viewport.addEventListener('touchmove', (e) => {
          if (!this.isPanning || e.touches.length !== 1) return;
          const delta = e.touches[0].clientX - this.panTouchStartX;
          this.setPan(this.panStartOffset + delta);
        }, { passive: true });

        this.viewport.addEventListener('touchend', () => {
          this.isPanning = false;
        });
      }

      setPan(val) {
        const minVal = -(3400 - window.innerWidth);
        const maxVal = 0;
        this.panOffsetX = Math.min(Math.max(val, minVal), maxVal);
        this.applyPanTransform();
      }

      applyPanTransform() {
        this.strip.style.transform = `translateX(${this.panOffsetX}px)`;
      }

paintBackgroundPanorama() {
    const w = 3400;
    const h = window.innerHeight;

    this.canvas.width = w;
    this.canvas.height = h;

    const ctx = this.ctx;

    // Fondo de seguridad
    ctx.fillStyle = "#050505";
    ctx.fillRect(0, 0, w, h);

    // Fotografías de NASA
    const imageURL = this.currentBody === 'moon'
       ? 'https://images-assets.nasa.gov/image/as08-14-2383/as08-14-2383~orig.jpg'
       : 'https://images-assets.nasa.gov/image/PIA05049/PIA05049~orig.jpg';

    const img = new Image();

    img.onload = () => {

       // Evitar dibujar una imagen de otro planeta
       if (!this.canvas || !this.ctx) return;

       const canvasRatio = w / h;
       const imageRatio = img.width / img.height;

       let sx = 0;
       let sy = 0;
       let sw = img.width;
       let sh = img.height;

       // Recortar sin deformar la fotografía
       if (imageRatio > canvasRatio) {
            sw = img.height * canvasRatio;
            sx = (img.width - sw) / 2;
        } else {
            sh = img.width / canvasRatio;
            sy = (img.height - sh) / 2;
        }

        ctx.clearRect(0, 0, w, h);

        ctx.drawImage(
            img,
            sx, sy, sw, sh,
            0, 0, w, h
        );

        // Oscurecimiento suave para mantener visibles los elementos
        const gradient = ctx.createLinearGradient(0, 0, 0, h);

        gradient.addColorStop(0, "rgba(0,0,0,0.05)");
        gradient.addColorStop(1, "rgba(0,0,0,0.25)");

        ctx.fillStyle = gradient;
        ctx.fillRect(0, 0, w, h);
    };

    img.onerror = () => {
        console.error("Error al cargar el panorama:", imageURL);
    };

    img.src = imageURL;
}
      generateHotspots() {

          pin.className = 'archaeo-hotspot';
          pin.style.left = `${item.x}px`;
          const topY = item.yRatio ? Math.round(h * item.yRatio) : item.y;
          pin.style.top = `${topY}px`;

          pin.innerHTML = `
            <div class="hotspot-radar-wave"></div>
            <div class="hotspot-center-dot"></div>
            <div class="hotspot-tag">${item.name}</div>
          `;

          pin.addEventListener('click', () => {
            sound.playBeaconPing();
            this.openTechnicalInspection(item);
          });

          this.hotspotsLayer.appendChild(pin);
        });
      }
